import requests

API_KEY = "YOUR_API_KEY"
SERVER_URL = "https://YOUR_SERVER"

def create_license(slug, key, discord_id=None, note=None):
    res = requests.post(
        f"{SERVER_URL}/api/licenses/create",
        headers={"X-API-Key": API_KEY, "Content-Type": "application/json"},
        json={"slug": slug, "license_key": key, "discord_id": discord_id, "note": note}
    )
    return res.json()
const axios = require('axios');

async function createLicense(slug, key, discordId, note) {
  const res = await axios.post('https://YOUR_SERVER/api/licenses/create', {
    slug,
    license_key: key,
    discord_id: discordId,
    note
  }, {
    headers: { 'X-API-Key': 'YOUR_API_KEY' }
  });
  return res.data;
}
